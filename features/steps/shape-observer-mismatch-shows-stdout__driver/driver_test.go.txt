	"github.com/KogenAI/kogen-coordinator/project"

// TestShapeObserverStdoutAcceptance re-runs the two controls of one declared observer through
// observerItems and writes the resulting items as JSON to $SOS_OUT. A positive
// $SOS_DEADLINE_MS shortens the probe deadline so that a sleeping control times out.
func TestShapeObserverStdoutAcceptance(t *testing.T) {
	raw, out := os.Getenv("SOS_CONTROLS"), os.Getenv("SOS_OUT")
	if raw == "" || out == "" {
		t.Skip("driven by the acceptance step")
	}
	if ms, err := strconv.Atoi(os.Getenv("SOS_DEADLINE_MS")); err == nil && ms > 0 {
		_, restore := verify.ContractProbeDeadline(time.Duration(ms)*time.Millisecond, nil)
		defer restore()
	}
	var controls []struct {
		Command string `json:"command"`
		Exit    int    `json:"exit"`
	}
	if err := json.Unmarshal([]byte(raw), &controls); err != nil || len(controls) != 2 {
		t.Fatalf("controls: %v", err)
	}
	root := t.TempDir()
	observer := "obs.js"
	if err := os.WriteFile(filepath.Join(root, observer), []byte("// observer\n"), 0o644); err != nil {
		t.Fatal(err)
	}
	checks := []map[string]any{}
	for i, name := range []string{"good", "bad"} {
		checks = append(checks, map[string]any{"name": name, "command": controls[i].Command, "exit": controls[i].Exit, "output": ""})
	}
	decl, _ := json.Marshal(map[string]any{"id": "obs", "kind": "observer", "inputs": []string{observer}, "observers": []string{observer}, "checks": checks})
	intent := &project.Intent{Acceptance: []string{observer}, Feasibility: []string{"- evidence: " + string(decl)}}
	env := append(os.Environ(), "SOS_SECRET=sos-secret-value-123456")
	ev, err := NewEvidence(filepath.Join(t.TempDir(), "ev"), "sos")
	if err != nil {
		t.Fatal(err)
	}
	sh := &shaping{s: sys.NewWith(env, root), root: root, ev: ev, rd: &verify.Readiness{Root: root}}
	items, err := sh.observerItems(context.Background(), intent)
	if err != nil {
		t.Fatal(err)
	}
	b, _ := json.Marshal(items)
	if err := os.WriteFile(out, b, 0o644); err != nil {
		t.Fatal(err)
	}
}
