package shape

// The acceptance harness of k-shape-data-stores-no-tool. Its acceptance step overlays it into
// ./shape, builds the shape test binary with `go test -c` and runs this test once per
// KSDSNT_MODE. KSDSNT_MODE plain and stores drive the native shaper
// (Shape) against the package's scripted local provider; external drives Shape with the
// package's scripted Claude Code shaper. It prints one KSDSNT observation line and judges
// nothing itself: the step compares the observations with the feature's expected values.

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"testing"

	"github.com/KogenAI/kogen-executor/executor"
	"kogen-loop/sys"
)

// The synthetic fixture: an account's memberships are rows of db/memberships.csv, which its
// Account model (lib/account.js) reaches, and one export file per account, exports/1.json and
// exports/2.json, written by lib/exports.js, which the model does not reach. lib/notes.txt holds
// only other keys, and .kogen/notes.txt is Kogen's own directory.
var ksdsntFixture = map[string]string{
	"lib/account.js": "// The Account model: an account's memberships are the rows of db/memberships.csv with its account_id.\n" +
		"const fs = require('node:fs');\n\n" +
		"exports.memberships = id => fs.readFileSync('db/memberships.csv', 'utf8').trim().split('\\n').slice(1)\n" +
		"  .filter(row => row.split(',')[0] === String(id));\n",
	"db/memberships.csv": "account_id,user\n1,ada\n2,bob\n",
	"lib/exports.js": "// Monthly exports: one file per account at exports/<account_id>.json. The Account model does not reach them.\n" +
		"const fs = require('node:fs');\n\n" +
		"exports.write = (id, data) => fs.writeFileSync(`exports/${id}.json`, JSON.stringify({ account_id: id, ...data }) + '\\n');\n",
	"exports/1.json":   "{\"account_id\":1,\"invoices\":3}\n",
	"exports/2.json":   "{\"account_id\":2,\"invoices\":1}\n",
	"lib/notes.txt":    "my_account_id and account_ids are other keys\n",
	".kogen/notes.txt": "account_id\n",
}

const ksdsntRequest = "Purge a cancelled account's data: `node lib/purge.js <id>` removes what the repository keeps for that account.\n"

func ksdsntIntent(decisions, stores []string) string {
	return "# Purge a cancelled account's data\n\nSlug: is-even\nModel: not applicable — one command removes rows and files; no state machine\nAcceptance:\n" +
		"- .kogen/intents/is-even/acceptance.feature\n- features/steps/is_even_steps.js\n\n" +
		"`node lib/purge.js <id>` removes what the repository keeps for a cancelled account.\n\n" +
		"## Decisions\n" + strings.Join(decisions, "\n") + "\n\n" +
		"## Data stores\n" + strings.Join(stores, "\n") + "\n\n" +
		"## Source comparison\n- addressed: purge a cancelled account's data; source request; contract acceptance.feature.\n\n" +
		"## Tests\n- Purging removes the account's data: the purge scenario (scenario)\n\n## Feasibility\n- none\n"
}

const ksdsntModelDecision = "- `lib/account.js` is the Account model's code, not a store; the purge removes the memberships rows it reads (source: request)"

// Package A is scoped to what the Account model reaches; package B covers every file that
// references account_id.
var (
	ksdsntIntentA = ksdsntIntent(
		[]string{"- Purging an account removes its memberships rows (source: request)", ksdsntModelDecision},
		[]string{"- key `account_id`", "- `db/memberships.csv`: in scope, observed by `db/memberships.csv`", "- `lib/account.js`: excluded by decision 2"})
	ksdsntIntentB = ksdsntIntent(
		[]string{"- Purging an account removes its memberships rows and its export file (source: request)", ksdsntModelDecision,
			"- `lib/exports.js` is the code that writes the export files, not a store; the purge removes the files it wrote (source: request)"},
		[]string{"- key `account_id`", "- `db/memberships.csv`: in scope, observed by `db/memberships.csv`", "- `exports/`: in scope, observed by `exports/`",
			"- `lib/account.js`: excluded by decision 2", "- `lib/exports.js`: excluded by decision 3"})
	ksdsntStepsA = shpSteps + "// Observes db/memberships.csv.\n"
	ksdsntStepsB = shpSteps + "// Observes db/memberships.csv and exports/.\n"
)

func ksdsntPackage(intent, steps string) map[string]any {
	return map[string]any{"files": []any{
		map[string]any{"path": ".kogen/intents/is-even/intent.md", "content": intent},
		map[string]any{"path": ".kogen/intents/is-even/acceptance.feature", "content": shpFeature},
		map[string]any{"path": "features/steps/is_even_steps.js", "content": steps},
	}}
}

func TestKShapeDataStoresNoToolAcceptance(t *testing.T) {
	mode := os.Getenv("KSDSNT_MODE")
	var f *shpFx
	var x *shapeExec
	request, wantIntent := "", ""
	switch mode {
	case "plain":
		f = newShpFx(t, shpSubmit("c1", shpFiles()))
		request, wantIntent = "Add calc.is_even(n). Non-integers raise TypeError.", shpIntentMD
	case "stores":
		f = newShpFx(t,
			shpAnswer(shpCall("ds1", "data_stores", map[string]any{"keys": []any{"account_id"}})),
			shpSubmit("c1", ksdsntPackage(ksdsntIntentA, ksdsntStepsA)),
			shpSubmit("c2", ksdsntPackage(ksdsntIntentB, ksdsntStepsB)))
		for p, c := range ksdsntFixture {
			f.write(p, c)
		}
		f.git("add", "-A")
		f.git("commit", "-qm", "Add the account fixture")
		request, wantIntent = ksdsntRequest, ksdsntIntentB
	case "external":
		f = newShpFx(t)
		f.useExternalShape(6)
		for p, c := range ksdsntFixture {
			f.write(p, c)
		}
		f.git("add", "-A")
		f.git("commit", "-qm", "Add the account fixture")
		x = &shapeExec{t: t, rounds: []func(int, string) (executor.RoundResult, error){
			result(ksdsntPackage(ksdsntIntentA, ksdsntStepsA)), result(ksdsntPackage(ksdsntIntentB, ksdsntStepsB))}}
		request, wantIntent = ksdsntRequest, ksdsntIntentB
	default:
		t.Fatalf("KSDSNT_MODE must be plain, stores or external, not %q", mode)
	}
	f.setReport("PASSED", 0)
	var exit int
	var err error
	if x != nil {
		exit, err = f.shapeWith(x, request)
	} else {
		exit, err = f.shape(request)
	}

	obs := map[string]any{"mode": mode, "exit": exit, "error": ""}
	if err != nil {
		obs["error"] = err.Error()
		if e, ok := sys.AsError(err); ok {
			obs["exit"] = e.Exit
		}
	}
	reqs := f.Prov.requests()
	obs["shaper_requests"] = len(reqs)
	if len(reqs) > 0 {
		ins, _ := reqs[0]["instructions"].(string)
		sum := sha256.Sum256([]byte(ins))
		obs["instructions_sha256"] = hex.EncodeToString(sum[:])
		obs["instructions_mention_data_stores"] = strings.Contains(ins, "Data stores") || strings.Contains(ins, "data_stores")
		obs["tools"] = strings.Join(shpToolNames(reqs[0]), ", ")
		obs["data_stores_tool"] = nil
		list, _ := reqs[0]["tools"].([]any)
		for _, d := range list {
			if m, _ := d.(map[string]any); m != nil && m["name"] == "data_stores" {
				obs["data_stores_tool"] = m
			}
		}
	}
	if x != nil {
		obs["shaper_sessions"] = len(x.starts)
		obs["shaper_rounds"] = x.calls
		obs["external_data_stores_paragraph"] = nil
		if len(x.starts) > 0 {
			for _, line := range strings.Split(x.starts[0].Brief.Stable, "\n") {
				if strings.HasPrefix(line, "Data stores: ") {
					obs["external_data_stores_paragraph"] = line
					break
				}
			}
		}
		obs["second_round_feedback"] = nil
		if len(x.inputs) > 1 {
			obs["second_round_feedback"] = strings.SplitN(x.inputs[1].Feedback, "\n\nCumulative proposal manifest: ", 2)[0]
		}
	}
	if mode == "stores" {
		obs["data_stores_output"] = nil
		obs["first_submission_feedback"] = nil
		if len(reqs) > 1 {
			if out, ok := shpOutputs(reqs[1])["ds1"]; ok {
				obs["data_stores_output"] = out
			}
		}
		if len(reqs) > 2 {
			if out, ok := shpOutputs(reqs[2])["c1"]; ok {
				obs["first_submission_feedback"] = out
			}
		}
	}
	gates := []any{}
	noTool := any(nil)
	if b, rerr := os.ReadFile(filepath.Join(f.Root, ".kogen/local/shaping/is-even/events.ndjson")); rerr == nil {
		for _, line := range strings.Split(string(b), "\n") {
			var ev map[string]any
			if line == "" || json.Unmarshal([]byte(line), &ev) != nil {
				continue
			}
			switch ev["event"] {
			case "submission_refused":
				gates = append(gates, ev["gate"])
			case "shaping_turn_counts":
				noTool = ev["no_tool_turns"]
			}
		}
	}
	obs["refused_gates"] = gates
	obs["no_tool_turns"] = noTool
	obs["acceptance_runs"] = f.runs()
	published := false
	if b, rerr := os.ReadFile(filepath.Join(f.Root, ".kogen/intents/is-even/intent.md")); rerr == nil {
		published = string(b) == wantIntent && f.exists(".kogen/local/readiness/is-even.json")
	}
	obs["published"] = published
	last := ""
	if lines := strings.Split(strings.TrimSuffix(f.Stdout.String(), "\n"), "\n"); len(lines) > 0 {
		last = lines[len(lines)-1]
	}
	obs["last_stdout_line"] = last
	b, jerr := json.Marshal(obs)
	if jerr != nil {
		t.Fatal(jerr)
	}
	fmt.Println("KSDSNT " + string(b))
}
