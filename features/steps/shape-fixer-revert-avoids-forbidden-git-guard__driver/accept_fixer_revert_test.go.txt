func accfrEnv(path string) []string {
	var env []string
	for _, kv := range os.Environ() {
		if strings.HasPrefix(kv, "GIT_") || (path != "" && strings.HasPrefix(kv, "PATH=")) {
			continue
		}
		env = append(env, kv)
	}
	env = append(env, "GIT_CONFIG_GLOBAL=/dev/null", "GIT_CONFIG_NOSYSTEM=1", "GIT_TERMINAL_PROMPT=0")
	if path != "" {
		env = append(env, "PATH="+path)
	}
	return env
}

func accfrGit(t *testing.T, root string, env []string, args ...string) string {
	t.Helper()
	cmd := exec.Command("git", args...)
	cmd.Dir = root
	cmd.Env = env
	out, err := cmd.CombinedOutput()
	if err != nil {
		t.Fatalf("git %s: %v\n%s", strings.Join(args, " "), err, out)
	}
	return string(out)
}

func accfrWrite(t *testing.T, root, rel, content string, mode os.FileMode) {
	t.Helper()
	p := filepath.Join(root, rel)
	if err := os.MkdirAll(filepath.Dir(p), 0o755); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(p, []byte(content), mode); err != nil {
		t.Fatal(err)
	}
	if err := os.Chmod(p, mode); err != nil {
		t.Fatal(err)
	}
}

func accfrRead(t *testing.T, root, rel string) string {
	t.Helper()
	b, err := os.ReadFile(filepath.Join(root, rel))
	if err != nil {
		t.Fatal(err)
	}
	return string(b)
}

// accfrFixture is a committed project: README.md, shaped.txt, an executable bin/run.sh and a link lnk to README.md.
func accfrFixture(t *testing.T) (string, []string) {
	t.Helper()
	root, env := t.TempDir(), accfrEnv("")
	accfrGit(t, root, env, "init", "-q")
	accfrGit(t, root, env, "config", "user.name", "Fixture")
	accfrGit(t, root, env, "config", "user.email", "fixture@example.invalid")
	accfrGit(t, root, env, "config", "commit.gpgsign", "false")
	accfrGit(t, root, env, "config", "core.hooksPath", t.TempDir())
	accfrWrite(t, root, "README.md", "# Calc\n\nA tiny calculator.\n", 0o644)
	accfrWrite(t, root, "shaped.txt", "shaped\n", 0o644)
	accfrWrite(t, root, "bin/run.sh", "#!/bin/sh\necho hi\n", 0o755)
	if err := os.Symlink("README.md", filepath.Join(root, "lnk")); err != nil {
		t.Fatal(err)
	}
	accfrGit(t, root, env, "add", "-A")
	accfrGit(t, root, env, "commit", "-qm", "fixture")
	return root, env
}

func accfrPass(root string, env []string, fix string, shaped ...string) error {
	sh := &shaping{s: sys.NewWith(env, root), repo: git.New(root, env), root: root,
		cfg: &project.Config{Fix: []string{fix}}, ev: &Evidence{}, started: time.Now()}
	return sh.fixAttempt(context.Background(), shaped)
}

func TestAccFixerRevertTrackedFile(t *testing.T) {
	root, env := accfrFixture(t)
	head := accfrRead(t, root, "README.md")
	if err := accfrPass(root, env, "echo more >> README.md; git add README.md; echo formatted >> shaped.txt", "shaped.txt"); err != nil {
		t.Fatalf("the pass returned an error: %v", err)
	}
	if got := accfrRead(t, root, "README.md"); got != head {
		t.Fatalf("README.md = %q, want the HEAD content %q", got, head)
	}
	if staged := accfrGit(t, root, env, "diff", "--cached", "--name-only"); staged != "" {
		t.Fatalf("staged paths = %q, want none", staged)
	}
	if status := accfrGit(t, root, env, "status", "--porcelain"); status != " M shaped.txt\n" {
		t.Fatalf("status = %q, want only the shaped file", status)
	}
	if got := accfrRead(t, root, "shaped.txt"); got != "shaped\nformatted\n" {
		t.Fatalf("shaped.txt = %q, want its formatting kept", got)
	}
}

func TestAccFixerRevertModeAndLink(t *testing.T) {
	root, env := accfrFixture(t)
	head := accfrRead(t, root, "bin/run.sh")
	fix := "echo x >> bin/run.sh; chmod 644 bin/run.sh; rm lnk; echo plain > lnk; echo formatted >> shaped.txt"
	if err := accfrPass(root, env, fix, "shaped.txt"); err != nil {
		t.Fatalf("the pass returned an error: %v", err)
	}
	if got := accfrRead(t, root, "bin/run.sh"); got != head {
		t.Fatalf("bin/run.sh = %q, want %q", got, head)
	}
	st, err := os.Lstat(filepath.Join(root, "bin/run.sh"))
	if err != nil || st.Mode()&0o100 == 0 {
		t.Fatalf("bin/run.sh lost its executable bit: %v %v", st, err)
	}
	if target, err := os.Readlink(filepath.Join(root, "lnk")); err != nil || target != "README.md" {
		t.Fatalf("lnk is not a link to README.md: target %q, err %v", target, err)
	}
	if status := accfrGit(t, root, env, "status", "--porcelain"); status != " M shaped.txt\n" {
		t.Fatalf("status = %q, want only the shaped file", status)
	}
	if staged := accfrGit(t, root, env, "ls-files", "-s", "--", "bin/run.sh", "lnk"); !strings.HasPrefix(staged, "100755 ") || !strings.Contains(staged, "\n120000 ") {
		t.Fatalf("index entries = %q, want 100755 and 120000", staged)
	}
}

func TestAccFixerRevertNoForbiddenCommand(t *testing.T) {
	root, env := accfrFixture(t)
	real, err := exec.LookPath("git")
	if err != nil {
		t.Fatal(err)
	}
	bin, log := t.TempDir(), filepath.Join(t.TempDir(), "git.log")
	shim := "#!/bin/sh\nprintf '%s\\n' \"$*\" >> '" + log + "'\nexec '" + real + "' \"$@\"\n"
	accfrWrite(t, bin, "git", shim, 0o755)
	shimEnv := accfrEnv(bin + ":" + os.Getenv("PATH"))

	if err := accfrPass(root, shimEnv, "echo more >> README.md; echo x >> bin/run.sh; echo formatted >> shaped.txt", "shaped.txt"); err != nil {
		t.Fatalf("the pass returned an error: %v", err)
	}
	logged, err := os.ReadFile(log)
	if err != nil || len(logged) == 0 {
		t.Fatalf("the git shim saw no command: %v", err)
	}
	forbidden := map[string]bool{"restore": true, "reset": true, "checkout": true, "clean": true, "symbolic-ref": true, "worktree": true, "remote": true}
	for _, line := range strings.Split(strings.TrimSpace(string(logged)), "\n") {
		args := strings.Fields(line)
		for i := 0; i < len(args); i++ {
			a := args[i]
			if a == "-c" || a == "-C" || a == "--git-dir" || a == "--work-tree" {
				i++
				continue
			}
			if strings.HasPrefix(a, "-") {
				continue
			}
			if forbidden[a] {
				t.Fatalf("the pass ran a forbidden git command: %q", line)
			}
			break
		}
	}
	if status := accfrGit(t, root, env, "status", "--porcelain"); status != " M shaped.txt\n" {
		t.Fatalf("status = %q, want only the shaped file", status)
	}

	// The guard of this module version is the one that ships: it refuses restore.
	if _, err := git.New(root, env).Run("restore", "--", "README.md"); !errors.Is(err, git.ErrForbidden) {
		t.Fatalf("restore through the guarded runner: err = %v, want git.ErrForbidden", err)
	}
}

func TestAccFixerRevertSnapshotted(t *testing.T) {
	root, env := accfrFixture(t)
	mine := accfrRead(t, root, "README.md") + "mine\n"
	accfrWrite(t, root, "README.md", mine, 0o644)
	if err := accfrPass(root, env, "echo fixer >> README.md; echo junk > junk.txt; echo formatted >> shaped.txt", "shaped.txt"); err != nil {
		t.Fatalf("the pass returned an error: %v", err)
	}
	if got := accfrRead(t, root, "README.md"); got != mine {
		t.Fatalf("README.md = %q, want the pre-fixer text %q", got, mine)
	}
	if _, err := os.Lstat(filepath.Join(root, "junk.txt")); err == nil {
		t.Fatal("junk.txt, which did not exist before the fixers, was kept")
	}
}
