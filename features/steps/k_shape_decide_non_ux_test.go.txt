package shape

// The acceptance harness of k-shape-decide-non-ux. Its acceptance step overlays it into ./shape,
// builds the shape test binary with `go test -c` and runs this test once per KSDNU_MODE.
// decides, asks-ui and environment drive the native shaper (Shape) against the package's scripted
// local provider; external drives Shape with the package's scripted Claude Code shaper. The
// fixtures are synthetic stand-ins for the private AoR task kits. It prints one KSDNU observation
// line and judges nothing itself: the step compares the observations with the feature's values.

)

// The synthetic billing fixture: accounts already carry a comped? stub, billing selects active
// and past-due accounts, and charges go through a gateway module rather than a command-line tool.
var ksdnuFixture = map[string]string{
	"app/models/account.rb": "class Account\n  attr_reader :status, :plan_price\n\n  # A comped account is never charged. Stub until the admin screen sets it.\n  def comped?\n    false\n  end\nend\n",
	"lib/billing.rb": "require_relative 'gateway'\n\n# Monthly billing: every active or past-due account is charged its plan price.\nmodule Billing\n  STATUSES = %w[active past_due].freeze\n\n" +
		"  def self.run(accounts)\n    accounts.select { |a| STATUSES.include?(a.status) }.each { |a| Gateway.charge(a, a.plan_price) }\n  end\nend\n",
	"lib/gateway.rb": "# The payment gateway is an HTTP client; billing needs no command-line tool.\nmodule Gateway\n  def self.charge(account, amount)\n    raise NotImplementedError\n  end\nend\n",
}

const ksdnuBillingRequest = "Monthly billing must not charge comped accounts: `bin/bill` charges every other billable account its plan price.\n"

const ksdnuReactionRequest = "Let users react to a comment with a short text reaction of at most 16 characters.\n"

const ksdnuProbe = "command -v stripe || echo no-stripe"

func ksdnuQ(text string, options ...string) map[string]any {
	list := []any{}
	for i, o := range options {
		list = append(list, map[string]any{"label": o, "recommended": i == 0})
	}
	return map[string]any{"question": text, "options": list}
}

var (
	ksdnuComped  = ksdnuQ("Is there already a way to mark an account as comped, or should billing add a comped flag?", "Add a comped column to accounts", "Reuse an existing marker")
	ksdnuPastDue = ksdnuQ("Should billing charge past-due accounts?", "Yes, like active accounts", "No, skip them")
	ksdnuStripe  = ksdnuQ("Is the Stripe CLI installed where billing runs?", "Assume it is installed", "Assume it is not")
	ksdnuKey     = ksdnuQ("Is a Stripe test API key available in this environment?", "Assume one is set", "Assume none is set")
	ksdnuLength  = ksdnuQ("When a reaction is longer than 16 characters, should the field stop accepting characters at 16, or accept them and show an error on submit?", "Stop accepting characters at 16", "Accept them and show an error on submit")
)

func ksdnuQuestions(qs ...map[string]any) map[string]any {
	list := make([]any, len(qs))
	for i, q := range qs {
		list[i] = q
	}
	return map[string]any{"questions": list}
}

// ksdnuCheck is a question check answer: one class per question, numbered in order.
func ksdnuCheck(id string, classes ...string) shpStep {
	items := []any{}
	for i, c := range classes {
		items = append(items, map[string]any{"number": i + 1, "class": c, "why": "synthetic " + c + " question"})
	}
	return shpAnswer(shpCall(id, "submit_question_check", map[string]any{"items": items}))
}

func ksdnuIntent(title, prose string, decisions ...string) string {
	return "# " + title + "\n\nSlug: is-even\nModel: not applicable — one command without state\nAcceptance:\n" +
		"- .kogen/intents/is-even/acceptance.feature\n- features/steps/is_even_steps.js\n\n" + prose + "\n\n" +
		"## Decisions\n" + strings.Join(decisions, "\n") + "\n\n" +
		"## Source comparison\n- addressed: the request; source request; contract acceptance.feature.\n\n" +
		"## Tests\n- The requested behavior: the acceptance scenarios (scenario)\n\n## Feasibility\n- none\n"
}

const (
	ksdnuDecComped  = "- Billing skips accounts whose existing Account#comped? stub returns true; no comped flag is added (evidence: app/models/account.rb) (source: request)"
	ksdnuDecPastDue = "- Past-due accounts are charged like active ones, as Billing::STATUSES already selects them (evidence: lib/billing.rb) (source: request)"
	ksdnuDecStripe  = "- Billing proceeds without the Stripe CLI: charges go through the Gateway module, and the probe found no need for a stripe command (evidence: lib/gateway.rb; ran `" + ksdnuProbe + "`) (source: request)"
)

var (
	ksdnuBillingTitle = "Skip comped accounts in monthly billing"
	ksdnuBillingProse = "`bin/bill` charges every billable account except comped ones."
	ksdnuIntentBad    = ksdnuIntent(ksdnuBillingTitle, ksdnuBillingProse,
		"- Billing skips accounts whose existing Account#comped? stub returns true; no comped flag is added (evidence: app/models/comp.rb) (source: request)",
		ksdnuDecPastDue,
		"- Billing proceeds without the Stripe CLI: charges go through the Gateway module (evidence: lib/gateway.rb; ran `stripe --version`) (source: request)")
	ksdnuIntentGood     = ksdnuIntent(ksdnuBillingTitle, ksdnuBillingProse, ksdnuDecComped, ksdnuDecPastDue, ksdnuDecStripe)
	ksdnuIntentExternal = ksdnuIntent(ksdnuBillingTitle, ksdnuBillingProse, ksdnuDecComped)
	ksdnuIntentReaction = ksdnuIntent("Let users react to a comment", "A comment takes a text reaction of at most 16 characters.",
		"- The reaction field stops accepting characters at 16 (source: answer 1)")
)

func ksdnuPackage(intent string) map[string]any {
	return map[string]any{"files": []any{
		map[string]any{"path": ".kogen/intents/is-even/intent.md", "content": intent},
		map[string]any{"path": ".kogen/intents/is-even/acceptance.feature", "content": shpFeature},
		map[string]any{"path": "features/steps/is_even_steps.js", "content": shpSteps},
	}}
}

func ksdnuCommitFixture(f *shpFx) {
	for p, c := range ksdnuFixture {
		f.write(p, c)
	}
	f.git("add", "-A")
	f.git("commit", "-qm", "Add the billing fixture")
}

// ksdnuIsCheck is whether a request is a question check: it offers submit_question_check.
func ksdnuIsCheck(req map[string]any) bool {
	for _, n := range shpToolNames(req) {
		if n == "submit_question_check" {
			return true
		}
	}
	return false
}

// ksdnuUserText is the text of the request's first user message.
func ksdnuUserText(req map[string]any) string {
	input, _ := req["input"].([]any)
	for _, it := range input {
		m, _ := it.(map[string]any)
		if m["role"] != "user" {
			continue
		}
		switch c := m["content"].(type) {
		case string:
			return c
		case []any:
			var b strings.Builder
			for _, blk := range c {
				if bm, _ := blk.(map[string]any); bm != nil {
					s, _ := bm["text"].(string)
					b.WriteString(s)
				}
			}
			return b.String()
		}
	}
	return ""
}

// ksdnuSubmitID are the call ids of the scripted shaper's submissions.
var ksdnuSubmitID = regexp.MustCompile(`^[qf][0-9]$`)

func ksdnuLastLine(s string) string {
	lines := strings.Split(strings.TrimSuffix(s, "\n"), "\n")
	return lines[len(lines)-1]
}

func ksdnuError(obs map[string]any, prefix string, exit int, err error) {
	obs[prefix+"exit"], obs[prefix+"error"], obs[prefix+"error_lines"] = exit, "", []string{}
	if err != nil {
		obs[prefix+"error"] = err.Error()
		if e, ok := sys.AsError(err); ok {
			obs[prefix+"exit"] = e.Exit
			if e.Lines != nil {
				obs[prefix+"error_lines"] = e.Lines
			}
		}
	}
}

func TestKShapeDecideNonUXAcceptance(t *testing.T) {
	mode := os.Getenv("KSDNU_MODE")
	obs := map[string]any{"mode": mode}
	var f *shpFx
	var x *shapeExec
	request, want := "", ""
	switch mode {
	case "decides":
		f = newShpFx(t,
			shpAnswer(shpCall("q1", "submit", ksdnuQuestions(ksdnuComped, ksdnuPastDue, ksdnuStripe))),
			ksdnuCheck("qc1", "repository_fact", "product_policy", "environment"),
			shpAnswer(shpCall("r1", "read_file", map[string]any{"path": "app/models/account.rb"}), shpCall("r2", "read_file", map[string]any{"path": "lib/billing.rb"}),
				shpCall("p1", "probe", map[string]any{"command": ksdnuProbe})),
			shpSubmit("f1", ksdnuPackage(ksdnuIntentBad)),
			shpSubmit("f2", ksdnuPackage(ksdnuIntentGood)))
		ksdnuCommitFixture(f)
		request, want = ksdnuBillingRequest, ksdnuIntentGood
	case "asks-ui":
		f = newShpFx(t,
			shpAnswer(shpCall("q1", "submit", ksdnuQuestions(ksdnuLength))),
			ksdnuCheck("qc1", "ui_ux"),
			shpSubmit("f1", ksdnuPackage(ksdnuIntentReaction)))
		request, want = ksdnuReactionRequest, ksdnuIntentReaction
	case "environment":
		f = newShpFx(t,
			shpAnswer(shpCall("q1", "submit", ksdnuQuestions(ksdnuKey))),
			ksdnuCheck("qc1", "environment"),
			shpAnswer(shpCall("q2", "submit", ksdnuQuestions(ksdnuKey))),
			ksdnuCheck("qc2", "environment"))
		ksdnuCommitFixture(f)
		request = ksdnuBillingRequest
	case "external":
		f = newShpFx(t)
		f.useExternalShape(6)
		ksdnuCommitFixture(f)
		x = &shapeExec{t: t, rounds: []func(int, string) (executor.RoundResult, error){
			result(ksdnuQuestions(ksdnuComped)),
			result(map[string]any{"items": []any{map[string]any{"number": 1, "class": "repository_fact", "why": "synthetic repository_fact question"}}}),
			result(ksdnuPackage(ksdnuIntentExternal))}}
		request, want = ksdnuBillingRequest, ksdnuIntentExternal
	default:
		t.Fatalf("KSDNU_MODE must be decides, asks-ui, environment or external, not %q", mode)
	}
	f.setReport("PASSED", 0)

	var exit int
	var err error
	if x != nil {
		exit, err = f.shapeWith(x, request)
	} else {
		exit, err = f.shape(request)
	}
	if mode == "asks-ui" {
		ksdnuError(obs, "first_", exit, err)
		obs["first_stdout"] = f.Stdout.String()
		obs["first_questions_pending"] = f.exists(".kogen/local/shaping/is-even/questions.json")
		exit, err = f.shape("all: recommended")
	}
	ksdnuError(obs, "", exit, err)

	var shaper, checks []map[string]any
	for _, r := range f.Prov.requests() {
		if ksdnuIsCheck(r) {
			checks = append(checks, r)
		} else {
			shaper = append(shaper, r)
		}
	}
	obs["shaper_requests"] = len(shaper)
	obs["question_check_requests"] = len(checks)
	if len(shaper) > 0 {
		ins, _ := shaper[0]["instructions"].(string)
		sum := sha256.Sum256([]byte(ins))
		obs["instructions_sha256"] = hex.EncodeToString(sum[:])
		obs["tools"] = strings.Join(shpToolNames(shaper[0]), ", ")
	}
	if len(checks) > 0 {
		obs["question_check_tools"] = strings.Join(shpToolNames(checks[0]), ", ")
		list, _ := checks[0]["tools"].([]any)
		if len(list) > 0 {
			obs["question_check_tool"] = list[0]
		}
		obs["question_check_instructions"], _ = checks[0]["instructions"].(string)
		obs["question_check_input"] = ksdnuUserText(checks[0])
	}
	for _, r := range shaper {
		for id, out := range shpOutputs(r) {
			if ksdnuSubmitID.MatchString(id) {
				obs["feedback_"+id] = out
			}
		}
	}
	if x != nil {
		sessions, schemas := 0, 0
		for _, s := range x.starts {
			if strings.Contains(string(s.Schema), `"ui_ux"`) {
				schemas++
				obs["question_check_stable_first_line"] = strings.SplitN(s.Brief.Stable, "\n", 2)[0]
			} else {
				sessions++
			}
		}
		obs["shaper_sessions"], obs["question_check_sessions"], obs["rounds"] = sessions, schemas, x.calls
		for i, in := range x.inputs {
			obs[fmt.Sprintf("round_%d_feedback", i+1)] = strings.SplitN(in.Feedback, "\n\nCumulative proposal manifest: ", 2)[0]
		}
	}

	gates, classes := []any{}, []any{}
	if b, rerr := os.ReadFile(filepath.Join(f.Root, ".kogen/local/shaping/is-even/events.ndjson")); rerr == nil {
		for _, line := range strings.Split(string(b), "\n") {
			var ev map[string]any
			if line == "" || json.Unmarshal([]byte(line), &ev) != nil {
				continue
			}
			switch ev["event"] {
			case "submission_refused":
				gates = append(gates, ev["gate"])
			case "question_check":
				var cs []any
				items, _ := ev["items"].([]any)
				for _, it := range items {
					if m, _ := it.(map[string]any); m != nil {
						cs = append(cs, m["class"])
					}
				}
				classes = append(classes, cs)
			}
		}
	}
	obs["refused_gates"], obs["question_check_classes"] = gates, classes
	obs["questions_pending"] = f.exists(".kogen/local/shaping/is-even/questions.json")
	published := false
	if b, rerr := os.ReadFile(filepath.Join(f.Root, ".kogen/intents/is-even/intent.md")); rerr == nil {
		published = want != "" && string(b) == want && f.exists(".kogen/local/readiness/is-even.json")
	}
	obs["published"] = published
	obs["intent_exists"] = f.exists(".kogen/intents/is-even/intent.md")
	obs["last_stdout_line"] = ksdnuLastLine(f.Stdout.String())
	b, jerr := json.Marshal(obs)
	if jerr != nil {
		t.Fatal(jerr)
	}
	fmt.Println("KSDNU " + string(b))
}
